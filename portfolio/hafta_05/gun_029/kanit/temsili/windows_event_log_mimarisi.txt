2. STANDART WINDOWS OLAY YAPISI VE ALAN ANATOMİSİ (XML / EVTX)
----------------------------------------------------------------------------------------

 <Event xmlns="http://schemas.microsoft.com/win/2004/08/events/event">
   <System>
     <Provider Name="Service Control Manager" Guid="{...}"/> -> Olayı Üreten Bileşen
     <EventID>7045</EventID>                                 -> Olay Türü ve Kimliği
     <Level>4</Level>                                        -> Bilgi (Information)
     <TimeCreated SystemTime="2026-10-27T07:22:15.891Z"/>    -> UTC Zaman Damgası
     <EventRecordID>104205</EventRecordID>                   -> Sıralı Adli Kayıt No
     <Channel>System</Channel>                               -> Günlük Kanalı
     <Computer>SEC-CORP-WS01</Computer>                      -> Kaynak Ana Makine
     <Security UserID="S-1-5-18"/>                           -> İşlemi Yapan SID
   </System>
   <EventData>
     <Data Name="ServiceName">SysmonAgentService</Data>      -> Özel Yük Verileri
     <Data Name="ImagePath">C:\Program Files\sysmon64.exe</Data>
   </EventData>
 </Event>
